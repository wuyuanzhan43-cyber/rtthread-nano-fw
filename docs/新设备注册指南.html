新设备注册指南

面向当前 board.c + board_config.h 单文件板级结构， 说明如何从“启用一个实例”到“新增一个完整设备类”的注册流程。 所有示例都按 mr-library 风格：静态对象、路径寻址、dev/drv/ops 分层。

注册模型

先记住框架的注册模型：设备对象、驱动描述和操作表三者分离， 注册表只保存 fw_dev_t*。

fw_dev_t 基类 / 注册表槽位
name · type · ref_count · ops · drv
fw_drv_t 驱动描述
type + ops + data
fw_dev_ops_t 设备类操作
open · read · write · ioctl · isr
fw_dev_register() 注册表："name" → fw_dev_t*
最小注册模板
真实设备通常把 drv 和私有数据放进自己的结构体； 下面只是最简形态，用来理解三个必填项。
最小模板
static fw_dev_t s_dev;

static const fw_drv_t s_drv = {
    .type = FW_DEV_TYPE_XXX,
    .ops  = &s_ops,
    .data = &s_dev,
};

fw_status_t my_register(void)
{
    fw_status_t result;

    result = fw_dev_register(&s_dev, "mydev",
                             FW_DEV_TYPE_XXX, &s_drv);
    if (result != FW_OK) return result;

    /* fw_dev_register 只初始化基类，ops 由类/驱动设置 */
    s_dev.ops = &s_ops;
    return FW_OK;
}

fw_dev_t 必须是具体设备结构体的第一个成员， 这样 FW_CONTAINER_OF(dev, my_device_t, dev) 才能从基类 反推出外层对象。设备对象和 drv 必须是静态/全局存储， 不能注册栈变量。

先判断你属于哪种情况

情况需要改什么是否写注册代码
启用一个已支持的实例
UART2 / SPI2 / I2C2		 只改 board_config.h 表 + Kconfig 不需要
在已有 I2C/SPI/GPIO 总线上挂一个具体器件
传感器 / Flash / 按键		 写器件驱动，调用总线 API；可选注册成命名设备 可选

方案 A：启用已有实例（不写注册代码）

例如启用 I2C2。

第 1 步：确认板级表

board_config.h 已经包含 I2C1/2/3 的默认接线：

board_config.h
#define BOARD_I2C_TABLE_INIT                     \
{                                               \
    { 1U, {1U, 6U},  {1U, 7U},  4U },            \
    { 2U, {1U, 10U}, {1U, 11U}, 4U },            \
    { 3U, {0U, 8U},  {2U, 9U},  4U },            \
}

如果实际接线不同，只改这张表；不要在其他 C 文件里重复引脚。

第 2 步：打开 Kconfig

profiles/<name>/.config
FW_USING_I2C2=y

也可以用 python tools/fwconfig.py menuconfig --profile f407ve 交互式修改。

第 3 步：生成并构建

命令行
python tools/fwconfig.py sync --all
python tools/fwconfig.py gen-keil --profile f407ve
cmake --build --preset f407-debug

board.c 的 build_i2c_hw() 会自动过滤到 I2C2， 并调用 drv_i2c_register_all()，不需要改任何 C 代码。

第 4 步：应用里使用

app
fw_dev_t *i2c2 = fw_dev_find("i2c2");
if (i2c2 == NULL) return FW_ENODEV;
注意
驱动目前只覆盖 UART1..3、SPI1..3、I2C1..3。增加 UART4 属于 “驱动扩展”，需要改 drv_serial.c 的静态槽位和 stm32f407_map.c，不是只改 Kconfig。

方案 B：在已有总线上挂一个具体器件

以 I2C 温度传感器为例，假设 7 位地址 0x76。 这里有两种做法，按复杂度递增。

B1. 最简单：直接调用总线 API（推荐先这样做）

如果只是一个固定地址的传感器、应用直接调用，完全不需要进 fw_dev 注册表。用 fw_i2c_mem_read 读写寄存器即可。

sensor.c
#include "fw/dev.h"
#include "fw/i2c.h"

static fw_dev_t *s_i2c;

fw_status_t sensor_init(void)
{
    uint8_t id;

    s_i2c = fw_dev_find("i2c1");
    if (s_i2c == NULL) return FW_ENODEV;

    if (fw_i2c_mem_read(s_i2c, 0x76U, 0xD0U, 1U,
                        &id, 1U, 100U) < 0)
        return FW_EIO;

    return id == 0x60U ? FW_OK : FW_ENODEV;
}

fw_status_t sensor_read_temperature(int16_t *value)
{
    uint8_t raw[2];

    if (fw_i2c_mem_read(s_i2c, 0x76U, 0xFAU, 1U,
                        raw, 2U, 100U) < 0)
        return FW_EIO;

    *value = (int16_t)(((uint16_t)raw[0] << 8) | raw[1]);
    return FW_OK;
}

B2. 想让它成为 fw_dev_find("sensor0") 可访问的设备

需要自己实现一个最小设备对象。下面按 6 步完成。

第 1 步：定义对象和 ops

fw/my_sensor.h
#include "fw/dev.h"
#include "fw/i2c.h"

typedef struct {
    fw_dev_t dev;       /* 必须第一个成员 */
    fw_drv_t drv;       /* 必须保持有效 */
    fw_dev_t *bus;      /* i2c1/i2c2... */
    uint16_t addr7;     /* 7 位从机地址 */
} my_sensor_t;

fw_status_t my_sensor_init(my_sensor_t *sensor, const char *name,
                           fw_dev_t *bus, uint16_t addr7);

第 2 步：实现 fw_dev_ops_t

fw/my_sensor.c
#include "fw/my_sensor.h"

static int sensor_read(fw_dev_t *dev, int off, void *buf, size_t size,
                       uint32_t timeout_ms)
{
    my_sensor_t *sensor = FW_CONTAINER_OF(dev, my_sensor_t, dev);
    int result;

    /* off 当作寄存器地址，例如读温度寄存器 0xFA */
    result = fw_i2c_mem_read(sensor->bus, sensor->addr7,
                             (uint16_t)off, 1U, buf, size, timeout_ms);
    if (result < 0) {
        fw_dev_note_error(dev);
        return result;
    }
    return result;
}

static int sensor_ioctl(fw_dev_t *dev, int off, int cmd, void *args)
{
    (void)dev;
    (void)off;
    (void)cmd;
    (void)args;
    return FW_ENOTSUP;
}

static const fw_dev_ops_t sensor_ops = {
    .read  = sensor_read,
    .ioctl = sensor_ioctl,
    /* open/close/write/isr 不需要就留 NULL */
};

第 3 步：注册

fw/my_sensor.c · 注册
fw_status_t my_sensor_init(my_sensor_t *sensor, const char *name,
                           fw_dev_t *bus, uint16_t addr7)
{
    fw_status_t result;

    if (sensor == NULL || bus == NULL) return FW_EINVAL;

    sensor->bus = bus;
    sensor->addr7 = addr7;

    sensor->drv.type = FW_DEV_TYPE_SENSOR;  /* 见第 4 步 */
    sensor->drv.ops = &sensor_ops;
    sensor->drv.data = sensor;

    result = fw_dev_register(&sensor->dev, name,
                             FW_DEV_TYPE_SENSOR, &sensor->drv);
    if (result != FW_OK) return result;

    /* fw_dev_register 只初始化基类，ops 由类/驱动设置 */
    sensor->dev.ops = &sensor_ops;
    return FW_OK;
}

第 4 步：在 fw/dev.h 增加类型令牌

fw/dev.h
enum fw_dev_type {
    FW_DEV_TYPE_NONE = 0,
    FW_DEV_TYPE_PIN = 1,
    FW_DEV_TYPE_SERIAL,
    FW_DEV_TYPE_SPI,
    FW_DEV_TYPE_DISPLAY,
    FW_DEV_TYPE_I2C,
    FW_DEV_TYPE_ADC,
    FW_DEV_TYPE_PWM,
    FW_DEV_TYPE_SENSOR   /* 新增，不能改动已有值 */
};
不要复用 FW_DEV_TYPE_NONE
注册函数会拒绝 FW_DEV_TYPE_NONE。新增类型时只追加， 不要修改已有枚举值，避免影响已经写进测试或生成的类型判断。

第 5 步：在板级初始化中注册

如果传感器是焊在板子上的，放在 board.c。

board_config.h
#define BOARD_SENSOR_I2C      1U
#define BOARD_SENSOR_ADDR7    0x76U
board.c
static my_sensor_t s_sensor;

static fw_status_t board_sensor_init(void)
{
    fw_dev_t *bus = fw_dev_find("i2c1");

    if (bus == NULL) return FW_ENOTREADY;
    return my_sensor_init(&s_sensor, "sensor0", bus,
                          BOARD_SENSOR_ADDR7);
}

在 board_devices_init() 中、board_i2c_init() 之后加：

board.c · board_devices_init
BOARD_RUN("i2c", board_i2c_init());
BOARD_RUN("sensor", board_sensor_init());

如果是应用业务传感器，也可以放在 main() 创建任务之前注册； 不要在调度器运行后无锁注册。

第 6 步：登记源文件和容量

  • project/sources.json：把 my_sensor.c 加入 core_sources 或 app_sources。
  • 设备表默认容量 FW_DEV_MAX=16；设备多了要在 Kconfig 里调大。
命令行
python tools/fwconfig.py check-manifest
python tools/fwconfig.py sync --all

方案 C：新增一个完整设备类（以 DAC 为例）

这是“新外设控制器”，需要 class + driver + board 三层。 以 DAC 为例，按文件从内到外说明。

5.1 框架类头文件

fw/dac.h
typedef struct {
    uint16_t value;
} fw_dac_config_t;

typedef struct {
    int (*configure)(fw_dev_t *dev, const fw_dac_config_t *config);
    int (*write)(fw_dev_t *dev, uint16_t value);
} fw_dac_ops_t;

typedef struct {
    fw_dev_t dev;           /* 第一个成员 */
    fw_dac_config_t config;
} fw_dac_t;

fw_status_t fw_dac_register(fw_dac_t *dac, const char *name,
                            const fw_drv_t *drv,
                            const fw_dac_config_t *config);
fw_dac_t *fw_dac_find(const char *name);
int fw_dac_write(fw_dev_t *dev, uint16_t value);

5.2 框架类实现

core/fw_dac.c
static int dac_dev_write(fw_dev_t *dev, int off, const void *buf,
                         size_t size, uint32_t timeout_ms)
{
    const fw_dac_ops_t *ops = (const fw_dac_ops_t *)dev->drv->ops;
    int result;

    (void)off;
    (void)size;
    (void)timeout_ms;

    result = ops->write(dev, *(const uint16_t *)buf);
    if (result < 0) fw_dev_note_error(dev);
    return result;
}

static const fw_dev_ops_t dac_dev_ops = {
    .write = dac_dev_write,
};

fw_status_t fw_dac_register(fw_dac_t *dac, const char *name,
                            const fw_drv_t *drv,
                            const fw_dac_config_t *config)
{
    fw_status_t result;

    if (dac == NULL || drv == NULL || config == NULL) return FW_EINVAL;
    if (drv->type != FW_DEV_TYPE_DAC || drv->ops == NULL)
        return FW_EINVAL;

    result = fw_dev_register(&dac->dev, name,
                             FW_DEV_TYPE_DAC, drv);
    if (result != FW_OK) return result;

    dac->dev.ops = &dac_dev_ops;
    dac->config = *config;
    return FW_OK;
}

5.3 芯片驱动头文件

drv_dac.h
typedef struct {
    uint8_t instance;
    uint8_t out_port;
    uint8_t out_pin;
} drv_dac_hw_t;

fw_status_t drv_dac_register_all(const drv_dac_hw_t *hw,
                                 size_t count);

5.4 芯片驱动实现

drv_dac.c
typedef struct {
    fw_dac_t base;
    DAC_HandleTypeDef hal;
    uint8_t instance;
    fw_drv_t drv;
} drv_dac_t;

static drv_dac_t dac1;

static int drv_dac_configure(fw_dev_t *dev,
                             const fw_dac_config_t *config);
static int drv_dac_write(fw_dev_t *dev, uint16_t value);

static const fw_dac_ops_t drv_dac_ops = {
    drv_dac_configure,
    drv_dac_write,
};

fw_status_t drv_dac_register_all(const drv_dac_hw_t *hw,
                                 size_t count)
{
    fw_dac_config_t config = { .value = 0 };

    /* 配置 GPIO、时钟、HAL_DAC_Init... */
    dac1.instance = hw[0].instance;

    dac1.drv.type = FW_DEV_TYPE_DAC;
    dac1.drv.ops  = &drv_dac_ops;
    dac1.drv.data = &dac1;

    return fw_dac_register(&dac1.base, "dac1",
                           &dac1.drv, &config);
}

5.5 板级接线

board_config.h
typedef struct {
    uint8_t instance;
    board_pin_t out;
} board_dac_t;

#define BOARD_DAC_TABLE_INIT       \
{                                 \
    { 1U, { 0U, 4U } }, /* PA4 */ \
}
board.c
static const board_dac_t s_dacs[] = BOARD_DAC_TABLE_INIT;
static drv_dac_hw_t s_dac_hw[sizeof(s_dacs) / sizeof(s_dacs[0])];

static size_t build_dac_hw(void)
{
    /* 表 -> drv_dac_hw_t，按 FW_USING_DAC1 过滤 */
}

static fw_status_t board_dac_init(void)
{
    size_t count = build_dac_hw();

    if (count == 0U) return FW_OK;
    return drv_dac_register_all(s_dac_hw, count);
}

board.c · board_devices_init
BOARD_RUN("dac", board_dac_init());

方案 D：同一个类的另一颗芯片

例如用 ST7789 替换 ST7735S。不需要新增 fw_dev_type， 只要实现该类的 ops，并调用该类的 register。

st7789.c
static const fw_display_ops_t st7789_ops = {
    st7789_hw_init,
    st7789_begin_write,
    st7789_write_pixels,
    st7789_end_write,
    st7789_set_backlight,
};

fw_status_t st7789_init(st7789_t *lcd, const char *name,
                        const st7789_config_t *config)
{
    fw_display_info_t info;

    info.width = config->width;
    info.height = config->height;
    info.bits_per_pixel = 16U;

    return fw_display_register(&lcd->base, name,
                               &st7789_ops, lcd, &info);
}

然后在 board.c 的 board_display_init() 里， 根据 board_config.h 的面板类型选择调用 st7735s_init() 或 st7789_init()。

注册时的硬性约束

  1. fw_dev_t 必须是第一个成员， 否则 FW_CONTAINER_OF 失效。
  2. 设备对象、fw_drv_t、ops 表必须是静态/全局的； 不能注册栈变量。
  3. 设备名唯一："sensor0" 和 "/dev/sensor0" 等价；最长 15 字符。
  4. drv->type 必须等于注册时的 type， 类的 register 会校验。
  5. 注册时机：fw_dev_register 设计为调度器启动前调用； 运行期注册需要调用者自己保证互斥，最简单是放在 main() 创建任务之前。
  6. dev->ops 要在注册成功后设置； fw_dev_register 只初始化基类字段。
  7. 错误路径：ops 返回负的 fw_status_t； 类层在转发失败时调用 fw_dev_note_error()。
  8. 容量：FW_DEV_MAX 默认 16； 每个 pin/serialN/spiN/i2cN/adc/pwm/lcd/自定义设备都占一个槽位。
  9. ISR 里的 ops 不可阻塞；串口/SPI 的中断回调 只做 ring/信号量操作。
  10. 不要在 platform 驱动里包含 board_config.h； 板级事实只通过 drv_*_hw_t 传入。
